pretty_dist_name <- function(dist) {
  switch(
    dist,
    weibull = "Weibull",
    lognormal = "Log-Normal",
    loglogistic = "Log-Logistic",
    exponential = "Exponential",
    str_to_title(dist)
  )
}

resolve_results_dir <- function(path) {
  candidates <- c(
    path,
    file.path(getwd(), path),
    file.path(dirname(getwd()), path),
    file.path(dirname(dirname(getwd())), path)
  )
  hit <- candidates[dir.exists(candidates)]
  if (length(hit) == 0) stop("Cannot find results_dir: ", path)
  normalizePath(hit[1], winslash = "/", mustWork = TRUE)
}

3 Coverage Across Tau

p_cov <- ggplot(
  method_tau_summary,
  aes(
    x = tau_quantile,
    y = coverage_mean,
    group = method_label
  )
) +
  geom_hline(yintercept = target_coverage, color = "gray35", linewidth = 0.5) +
  geom_line(linewidth = 0.8) +
  geom_point(
    data = method_tau_summary %>% filter(estimator_family == "plug-in"),
    aes(color = method_label, shape = method_label),
    size = 2.2
  ) +
  geom_point(
    data = method_tau_summary %>% filter(estimator_family == "conformal"),
    aes(color = method_label, shape = method_label),
    size = 2.8,
    stroke = 1.0
  ) +
  geom_errorbar(aes(ymin = coverage_ci_low, ymax = coverage_ci_high), width = 0.015, alpha = 0.85) +
  scale_color_manual(values = method_color_values, breaks = method_levels) +
  scale_shape_manual(values = method_shape_values, breaks = method_levels) +
  scale_x_continuous(breaks = sort(unique(method_tau_summary$tau_quantile))) +
  labs(
    title = "Coverage vs Administrative Horizon",
    x = "tau quantile used in the DGP",
    y = "Mean empirical coverage",
    color = "Method",
    shape = "Method"
  ) +
  theme_bw(base_size = 11)

focus_tau <- sort(unique(sample_df$tau_quantile))
focus_tau <- if (any(abs(focus_tau - 0.90) < 1e-8)) {
  0.90
} else {
  focus_tau[ceiling(length(focus_tau) / 2)]
}

direct_df <- sample_df %>%
  filter(abs(tau_quantile - focus_tau) < 1e-8)

p_direct <- ggplot(direct_df, aes(y = obs_rank)) +
  geom_segment(aes(x = pi_lower, xend = pi_upper, yend = obs_rank, color = covered_label), alpha = 0.9) +
