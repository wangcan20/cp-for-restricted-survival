resolve_results_dir <- function(path) {
  candidates <- c(
    path,
    file.path(getwd(), path),
    file.path(dirname(getwd()), path)
  )
  hit <- candidates[dir.exists(candidates)]
  if (length(hit) == 0) stop("Cannot find results_dir: ", path)
  normalizePath(hit[1], winslash = "/", mustWork = TRUE)
}

save_plot <- function(filename, p, width = 9, height = 5.5, dpi = 300) {
  out_path <- file.path(fig_dir, filename)
  ggsave(out_path, p, width = width, height = height, dpi = dpi)
  pdf_name <- sub("\\.[^.]+$", ".pdf", filename)
  ggsave(file.path(fig_dir, pdf_name), p, width = width, height = height)
}

5 Direct/Descriptive Visualization

This plot shows individual predicted intervals and true T* values for sampled test observations from rep 1, highlighting covered vs missed cases, with a dashed line at realized tau.

Code 
mid_cens <- sort(unique(sample_df$cens_target))
mid_cens <- mid_cens[ceiling(length(mid_cens) / 2)]
tau_levels <- sort(unique(sample_df$tau_quantile))
show_tau <- unique(c(min(tau_levels), tau_levels[ceiling(length(tau_levels) / 2)], max(tau_levels)))

showcase <- sample_df %>%
  filter(dgp_id == "base", abs(cens_target - mid_cens) < 1e-8, tau_quantile %in% show_tau) %>%
  group_by(setting_id, tau_quantile) %>%
  slice_sample(n = 140)

if ("eta_hat" %in% names(showcase)) {
  showcase <- showcase %>% arrange(eta_hat, .by_group = TRUE)
} else {
  showcase <- showcase %>% arrange(T_star, .by_group = TRUE)
}

showcase <- showcase %>%
  mutate(obs_rank = row_number(), covered_label = if_else(covered == 1, "Covered", "Missed")) %>%
  ungroup() %>%
  mutate(panel = paste0("tau_q=", formatC(tau_quantile, digits = 2, format = "f"),
                        ", tau≈", formatC(tau, digits = 2, format = "f")))

p_direct <- ggplot(showcase, aes(y = obs_rank)) +
  geom_segment(aes(x = pi_lower, xend = pi_upper, yend = obs_rank, color = covered_label), alpha = 0.9) +
  geom_point(aes(x = T_star, color = covered_label), size = 1.2) +
  geom_vline(aes(xintercept = tau), linetype = "dashed", color = "gray35") +
  facet_wrap(~panel, scales = "free_x") +
  scale_color_manual(values = c("Covered" = "#1b9e77", "Missed" = "#d95f02")) +
  labs(
    title = "Sample-Level Intervals and Coverage by tau Setting",
    x = "Time scale",
    y = "Observation rank (sorted by estimated risk x^T beta_hat)",
    color = "Coverage"
  ) +
  theme_bw(base_size = 11)

save_plot("direct_interval_coverage_examples.png", p_direct, width = 11, height = 5.5)
